## Documentation consistency check

`package.json` scripts are `build, typecheck, test, verify, clean, prepack`. The check
extracts every `pnpm [run] <name>` / `npm run <name>` reference **from code spans and fenced
blocks only** (prose is ignored, so sentences like "if pnpm is not on `PATH`" cannot produce
false positives), classifies pnpm subcommands (`exec`, `install`, `add`, `remove`, …) as
verbs rather than scripts, verifies every relative markdown link target exists on disk, and
verifies every `#` fragment against the target file's heading slugs.

Exact command (run from the project root; the script is written, executed, and removed in one
invocation because `/tmp` is per-command ephemeral in this sandbox):

```bash
cd /home/bruno/dev/projects/dsh-opencode-go-usage && cat > "$PWD/.doc-consistency-check.cjs" <<'EOF'
const fs = require('fs'), path = require('path')
const pkg = JSON.parse(fs.readFileSync('package.json', 'utf8'))
const scripts = new Set(Object.keys(pkg.scripts || {}))
const PNPM_VERBS = new Set(['exec','install','add','remove','rm','why','dlx','update','start'])
const FILES = ['README.md','docs/install.md','docs/testing.md']
const slug = h => h.trim().toLowerCase().replace(/[^\w\- ]+/g,'').replace(/ +/g,'-')
const headingsOf = f => { const s = new Set(); for (const m of fs.readFileSync(f,'utf8').matchAll(/^#{1,6} +(.+?)\s*$/gm)) s.add(slug(m[1])); return s }
let bad = 0, scriptRefs = 0, fileLinks = 0, anchors = 0
const cache = new Map()
for (const file of FILES) {
  const text = fs.readFileSync(file,'utf8')
  const code = []
  for (const m of text.matchAll(/```[^\n]*\n([\s\S]*?)```/g)) code.push(m[1])
  for (const m of text.matchAll(/`([^`\n]+)`/g)) code.push(m[1])
  const seen = new Set()
  for (const chunk of code) for (const m of chunk.matchAll(/(?:^|[\s(])((?:pnpm|npm) (?:run )?([a-z][a-z0-9:-]*))/g)) {
    const phrase = m[1], name = m[2]
    if (seen.has(phrase)) continue
    seen.add(phrase)
    if (/^npm run /.test(phrase)) { scriptRefs++; if (scripts.has(name)) console.log(`ok script   ${file}  "${phrase}"`); else { console.log(`MISSING     ${file}  "${phrase}"`); bad++ }; continue }
    if (name === 'run') continue
    if (PNPM_VERBS.has(name)) { console.log(`pnpm verb   ${file}  "${phrase}"`); continue }
    scriptRefs++
    if (scripts.has(name)) console.log(`ok script   ${file}  "${phrase}" -> package script "${name}"`)
    else { console.log(`MISSING     ${file}  "${phrase}"`); bad++ }
  }
  for (const m of text.matchAll(/\]\(([^)\s]+)\)/g)) {
    const target = m[1]
    if (/^[a-z][a-z0-9+.-]*:/i.test(target)) { console.log(`external    ${file}  ${target}`); continue }
    if (target.startsWith('#')) { anchors++; if (!cache.has(file)) cache.set(file, headingsOf(file)); if (cache.get(file).has(target.slice(1))) console.log(`ok anchor   ${file}  ${target}`); else { console.log(`MISSING     ${file}  ${target}`); bad++ }; continue }
    const [rel, fragment] = target.split('#')
    fileLinks++
    const abs = path.resolve(path.dirname(file), rel)
    if (!fs.existsSync(abs)) { console.log(`MISSING     ${file}  ${target} -> file not found`); bad++; continue }
    if (fragment !== undefined) { anchors++; if (!cache.has(abs)) cache.set(abs, headingsOf(abs)); if (cache.get(abs).has(fragment)) console.log(`ok link     ${file}  ${target}`); else { console.log(`MISSING     ${file}  ${target} -> anchor not found`); bad++ } }
    else console.log(`ok link     ${file}  ${target}`)
  }
}
console.log('---')
console.log(`package.json scripts: ${[...scripts].join(', ')}`)
console.log(`checks: ${scriptRefs} script refs, ${fileLinks} relative file links, ${anchors} anchors`)
console.log(bad === 0 ? 'RESULT: PASS (0 missing)' : `RESULT: FAIL (${bad} missing)`)
process.exit(bad === 0 ? 0 : 1)
EOF
node "$PWD/.doc-consistency-check.cjs"; echo "CHECK_EXIT=$?"; rm -f "$PWD/.doc-consistency-check.cjs"
```
